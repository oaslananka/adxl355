cmake_minimum_required(VERSION 3.14)
project(adxl355-cpp-consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(adxl355-cpp 0.1 CONFIG REQUIRED)

add_executable(adxl355_cpp_consumer main.cpp)
target_link_libraries(adxl355_cpp_consumer PRIVATE adxl355::cpp)

add_executable(adxl355_cpp_noexcept_consumer main_noexcept.cpp)
target_link_libraries(adxl355_cpp_noexcept_consumer PRIVATE adxl355::cpp_noexcept)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adxl355_cpp_noexcept_consumer PRIVATE -fno-exceptions -fno-rtti)
elseif(MSVC)
    target_compile_options(adxl355_cpp_noexcept_consumer PRIVATE /GR-)
endif()
